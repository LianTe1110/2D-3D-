# MPI 多平面图像架构 实施计划

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** 将单平面顶点位移渲染升级为 3 层 MPI 多平面架构, 彻底解决撕裂和遮挡问题

**Architecture:** 后端按深度 Otsu 切分生成 3 张 RGBA 层纹理 PNG → 上传 MinIO → 前端 3 个独立 mesh 按 Z 排序渲染 + 深度测试实现遮挡

**Tech Stack:** Python/OpenCV/NumPy (后端层纹理生成), React/Three.js/GLSL (前端 MPI 渲染)

**Spec 文档:** `docs/superpowers/specs/2026-06-17-mpi-design.md`

- [ ] **Step 1: 在 `DepthAnythingV2Engine` 类中添加 `_compute_layer_thresholds` 方法**

在 `_postprocess` 方法之后、`predict_depth` 方法之前插入:

```python
    def _compute_layer_thresholds(
        self, depth_uint8: np.ndarray, n_layers: int
    ) -> list[int]:
        """计算层边界阈值

        3层: 两级 Otsu → [0, t1, t2, 255]
        其他: 等间距 → [0, step, 2*step, ..., 255]
        """
        if n_layers == 3:
            _, binary1 = cv2.threshold(
                depth_uint8, 0, 255, cv2.THRESH_BINARY + cv2.THRESH_OTSU
            )
            dark_region = depth_uint8[binary1 == 0]
            if dark_region.size > 10:
                t1, _ = cv2.threshold(
                    dark_region, 0, 255, cv2.THRESH_BINARY + cv2.THRESH_OTSU
                )
                t1 = int(t1)
            else:
                t1 = 85
            bright_region = depth_uint8[binary1 == 255]
            if bright_region.size > 10:
                t2, _ = cv2.threshold(
                    bright_region, 0, 255, cv2.THRESH_BINARY + cv2.THRESH_OTSU
                )
                t2 = int(t2)
            else:
                t2 = 170
            if t1 >= t2:
                t1, t2 = min(t1, t2) // 2, max(t1, t2)
            return [0, t1, t2, 255]
        else:
            step = 256 // n_layers
            return [i * step for i in range(n_layers)] + [255]
```

- [ ] **Step 2: 添加 `_generate_mpi_layers` 方法**

在 `_compute_layer_thresholds` 之后插入:

```python
    def _generate_mpi_layers(
        self, depth_uint8: np.ndarray, original_image: Image.Image, n_layers: int = 3
    ) -> list[Image.Image]:
        """将原图按深度切分为 N 层 RGBA 纹理

        Args:
            depth_uint8: 后处理完成的 uint8 深度图
            original_image: 原始 RGB 图像
            n_layers: 层数 (默认 3)

        Returns:
            list of RGBA PIL Image, 按深度从远到近排序
        """
        thresholds = self._compute_layer_thresholds(depth_uint8, n_layers)
        original_rgba = np.array(original_image.convert("RGBA"))

        layers = []
        for i in range(n_layers):
            if i < n_layers - 1:
                mask = (
                    (depth_uint8 >= thresholds[i])
                    & (depth_uint8 < thresholds[i + 1])
                ).astype(np.uint8) * 255
            else:
                mask = (depth_uint8 >= thresholds[i]).astype(np.uint8) * 255

            # 边缘膨胀: 5x5核×3次, 确保视差移动时不露缝
            kernel = cv2.getStructuringElement(cv2.MORPH_ELLIPSE, (5, 5))
            mask = cv2.dilate(mask, kernel, iterations=3)

            layer = original_rgba.copy()
            layer[:, :, 3] = mask
            layers.append(Image.fromarray(layer))

        return layers
```

- [ ] **Step 3: 添加 MPILayers 和 LayerMesh 组件**

在 `DepthMesh` 组件之后插入:

```tsx
const LAYER_Z_POSITIONS = [-1.0, 0.0, 1.2]

const artStyleMap: Record<string, number> = {
  original: 0, anime: 1, oil_painting: 2, watercolor: 3, sketch: 4, cyberpunk: 5, vintage: 6,
}

function LayerMesh({ textureUrl, zPosition, layerIndex, artStyle }: {
  textureUrl: string; zPosition: number; layerIndex: number; artStyle: string
}) {
  const { uploadedImage } = useEditorStore()
  const texture = useMemo(() => {
    const tex = new THREE.TextureLoader().load(textureUrl)
    tex.colorSpace = THREE.NoColorSpace
    tex.minFilter = THREE.LinearFilter
    tex.magFilter = THREE.LinearFilter
    return tex
  }, [textureUrl])

  const geometry = useMemo(() => {
    const imgW = uploadedImage?.width || 4
    const imgH = uploadedImage?.height || 3
    const aspect = imgW / imgH
    return new THREE.PlaneGeometry(2.0 * aspect, 2.0, 1, 1)
  }, [uploadedImage])

  const material = useMemo(() => {
    const isForeground = layerIndex === 2
    return new THREE.ShaderMaterial({
      uniforms: {
        uTexture: { value: texture },
        uArtStyle: { value: artStyleMap[artStyle] ?? 0 },
        uTexelSize: { value: new THREE.Vector2(1/1024, 1/1024) },
      },
      vertexShader: LAYER_VERTEX_SHADER,
      fragmentShader: LAYER_FRAGMENT_SHADER,
      transparent: true,
      depthTest: true,
      depthWrite: isForeground,
      blending: THREE.CustomBlending,
      blendSrc: THREE.OneFactor,
      blendDst: THREE.OneMinusSrcAlphaFactor,
      toneMapped: false,
    })
  }, [texture, artStyle])

  // 纹理加载后更新 texelSize
  useEffect(() => {
    if (texture.image) {
      const w = texture.image.width || 1024
      const h = texture.image.height || 1024
      ;(material as THREE.ShaderMaterial).uniforms.uTexelSize.value.set(1.0 / w, 1.0 / h)
    } else {
      texture.addEventListener('load', () => {
        const w = texture.image?.width || 1024
        const h = texture.image?.height || 1024
        ;(material as THREE.ShaderMaterial).uniforms.uTexelSize.value.set(1.0 / w, 1.0 / h)
      })
    }
  }, [texture, material])

  return (
    <mesh
      geometry={geometry}
      material={material}
      position={[0, 0, zPosition]}
      renderOrder={layerIndex}
    />
  )
}

function MPILayers() {
  const { mpiLayerUrls, uploadedImage, styleQuality } = useEditorStore()
  if (!mpiLayerUrls || mpiLayerUrls.length === 0) return null

  return (
    <>
      {mpiLayerUrls.map((url, i) => (
        <LayerMesh
          key={i}
          textureUrl={url}
          zPosition={LAYER_Z_POSITIONS[i] ?? 0}
          layerIndex={i}
          artStyle={styleQuality.artStyle}
        />
      ))}
    </>
  )
}
```
